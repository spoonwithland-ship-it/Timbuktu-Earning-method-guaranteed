const DB_KEY = "tb_shared_v1";
function loadDB() { try { return JSON.parse(localStorage.getItem(DB_KEY)||"null"); } catch { return null; } }
function saveDB(d) { try { localStorage.setItem(DB_KEY, JSON.stringify(d)); } catch {} }
function initDB() {
  let d = loadDB();
  if (!d) {
    d = {
      users: { admin: { pass:"admin123", name:"Admin", role:"admin", unlockedDays:[], methodUnlocked:false } },
      codes: {}
    };
    saveDB(d);
  }
  return d;
}
function makeCode(str) {
  let h=0; for(let i=0;i<str.length;i++) h=(Math.imul(31,h)+str.charCodeAt(i))|0;
  return Math.abs(h).toString(36).toUpperCase().slice(0,6);
}

                  <div style={{background:ul?`linear-gradient(135deg,${C.gDark},${C.gLight})`:"#e8e8e8",color:ul?C.black:"#aaa",borderRadius:"50%",width:38,height:38,display:"flex",alignItems:"center",justifyContent:"center",fontWeight:900,fontSize:14,flexShrink:0,boxShadow:ul?"0 2px 8px #2db52d44":"none"}}>
                    {d.day}
                  </div>
                  <div style={{flex:1,minWidth:0}}>
                    <div style={{fontWeight:700,fontSize:13,color:ul?C.black:"#888"}}>DAY {d.day}: {d.title}</div>
                    <div style={{...s.mu,fontSize:11,overflow:"hidden",textOverflow:"ellipsis",whiteSpace:"nowrap"}}>{d.video}</div>
                  </div>
                  <div style={{textAlign:"right",flexShrink:0}}>
                    <div style={{fontSize:18}}>{ul?"🔓":"🔒"}</div>
                    <div style={{color:ul?C.green:"#bbb",fontWeight:700,fontSize:10}}>{d.earning}</div>
                  </div>
                </div>
              </div>
            );
          })}
        </div>
      </div>
    );
  }

  // ── METHOD ────────────────────────────────────────────────────
  if(pg==="method"&&user) return(
    <div style={s.app}>
      <div style={s.hdr}>
        <div style={s.hdrInner}>
          <div style={s.logoBox}><Crown size={28}/><span style={{...s.logoText,fontSize:16}}>TIMBUKTU</span></div>
          <button style={s.btnOut} onClick={()=>setPg("dash")}>← BACK</button>
        </div>
      </div>
      <div style={s.wrap}>
        <div style={{...s.gcard,color:"#fff"}}>
          <div style={{fontWeight:900,fontSize:15,marginBottom:14,textAlign:"center"}}>🔓 FULL METHOD – UNLOCKED</div>
          <pre style={{whiteSpace:"pre-wrap",fontSize:13,lineHeight:1.9,fontFamily:"inherit"}}>{METHOD}</pre>
        </div>
      </div>
    </div>
  );

  // ── DAY DETAIL ────────────────────────────────────────────────
  if(pg==="day"&&dayIdx!==null){
    const d=DAYS[dayIdx];
    const du=db.users[user.id]||{};
    return(
      <div style={s.app}>
        <Modal/>
        <div style={s.hdr}>
          <div style={s.hdrInner}>
            <div style={s.logoBox}><Crown size={26}/><span style={{...s.logoText,fontSize:15}}>DAY {d.day}</span></div>
            <button style={s.btnOut} onClick={()=>setPg("dash")}>← BACK</button>
          </div>
        </div>
        <div style={s.wrap}>
          <div style={s.gcardYellow}>
            <div style={{color:C.black,fontSize:16,fontWeight:900}}>📅 DAY {d.day}: {d.title}</div>
            <div style={{color:C.black,fontSize:12,opacity:0.75,marginBottom:9}}>{d.video}</div>
            <span style={{...s.tag,background:"rgba(0,0,0,0.12)",border:"none",color:C.black}}>👁 {d.views} views</span>
            <span style={{...s.tag,background:"rgba(0,0,0,0.15)",border:"none",color:C.black,fontWeight:900}}>💰 {d.earning}</span>
          </div>
          <div style={{fontWeight:700,fontSize:13,marginBottom:9,color:C.black}}>📋 STEPS:</div>
          {d.steps.map((step,i)=>(
            <div key={i} style={{...s.card,display:"flex",gap:11,padding:13}}>
              <div style={{background:`linear-gradient(135deg,${C.gDark},${C.gLight})`,color:C.black,width:26,height:26,borderRadius:"50%",display:"flex",alignItems:"center",justifyContent:"center",fontWeight:900,fontSize:12,flexShrink:0}}>{i+1}</div>
              <div style={{fontSize:13,lineHeight:1.75,color:C.text}}>{step}</div>
            </div>
          ))}
          <div style={s.gcard}>
            <div style={{color:"#c8ff00",fontWeight:700,marginBottom:5,fontSize:13}}>💡 PRO TIP:</div>
            <div style={{fontSize:13,lineHeight:1.75,color:"#fff"}}>{d.tip}</div>
          </div>
          <div style={{display:"flex",gap:9}}>
            {dayIdx>0&&(
              <button style={{...s.btn,background:"#e8f5e8",color:C.gDark,boxShadow:"none",border:`1.5px solid ${C.green}`}}
                onClick={()=>setDayIdx(p=>p-1)}>← DAY {d.day-1}</button>
            )}
            {dayIdx<9&&(
              <button style={s.btn} onClick={()=>{
                const nd=d.day+1;
                const ul2=db.users[user.id]?.unlockedDays?.includes(nd);
                if(ul2) setDayIdx(p=>p+1);
                else{setPg("dash");setTimeout(()=>openModal(`day_${nd}`),100);}
              }}>DAY {d.day+1} →</button>
            )}
          </div>
        </div>
      </div>
    );
  }

  return <div style={s.app}><div style={{padding:20}}>Loading...</div></div>;
}
